NAME
                JumpMsgArea -- Changes a users current message area by list order.

   SYNOPSIS
                JumpMsgArea <NEXT|PREV|FIRST|LAST> [IGNORESIG]

   FUNCTION

   INPUTS

   RESULT
                Old Message Area

   RC
                 0 -- Message Area Changed
                 5 -- No more message areas
                10 -- Bad arguments
                20 -- Fatal Error

   EXAMPLE
                /*
                ** Scans the message areas
                */

                fin = 0

                Write '"Scanning Message Areas Forwards...nn"'

                JumpMsgArea 'FIRST'
                if RC = 0 then do
                        do until fin = 1
                                Write '"%2030|) %2031|n"'

                                JumpMsgArea 'NEXT'
                                if RC ~= 0 then fin = 1
                                end
                        end

                Write '"nScanning Message Areas Backwards...nn"'

                fin = 0

                JumpMsgArea 'LAST'
                if RC = 0 then do
                        do until fin = 1
                                Write '"%2030|) %2031|n"'

                                JumpMsgArea 'PREV'
                                if RC ~= 0 then fin = 1
                                end
                        end
